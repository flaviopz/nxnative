# Switches the app carries for this game (Resources/dm/env.txt, the Mac run command's environment), which the build bakes in from the profile.
# A player's own ~/Library/Application Support/nxnative/0400486024DEC000/env.txt is read first and wins.
# Without them the host makes a Metal object for every texture in the game's texture tables on entering a scene (over 19 GB on a 16 GB Mac mini M4:
# "Insufficient Memory" from the GPU, a black screen, once a restart of the Mac).
DM_NVN_POOL_BUDGET_MB=32
DM_NVN_GMEM_CHUNKS=1
DM_NVN_POOL_PER_FRAME=4
DM_NVN_UNC_ANY_POOL=1
DM_NVN_KEEP_SHARED_RETIRED=1
DM_FPS_TARGET=30
